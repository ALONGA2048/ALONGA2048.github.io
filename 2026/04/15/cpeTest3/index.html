# uva. 706 LC-Display

# 題目:

題目想要你用'-'的符號進行數字的顯示做成如七段顯示器的顯示方式

Sample Input
2 12345
3 67890
0 0

Sample Output:

解題思路:

  • 要求對應的數字輸出,我們可以先將建構數字的構造先設定起來
  • 依照第一位數來分別加長他的長度
  • 查看他的輸入輸出條件(輸出0 0 則為EOF)

import sys

# 七段顯示器的線段定義 (0-9)
# 陣列順序：(頂, 左上, 右上, 中, 左下, 右下, 底)
# 1 代表該線段有墨水('-' 或 '|')，0 代表全空白
segments = {
    '0': (1, 1, 1, 0, 1, 1, 1),
    '1': (0, 0, 1, 0, 0, 1, 0),
    '2': (1, 0, 1, 1, 1, 0, 1),
    '3': (1, 0, 1, 1, 0, 1, 1),
    '4': (0, 1, 1, 1, 0, 1, 0),
    '5': (1, 1, 0, 1, 0, 1, 1),
    '6': (1, 1, 0, 1, 1, 1, 1),
    '7': (1, 0, 1, 0, 0, 1, 0),
    '8': (1, 1, 1, 1, 1, 1, 1),
    '9': (1, 1, 1, 1, 0, 1, 1)
}

def get_horizontal(is_on, s):
    """產生橫向線段：例如 ' -- ' 或 '    '"""
    char = "-" if is_on else " "
    return " " + (char * s) + " "

def get_vertical(left_on, right_on, s):
    """產生直向線段：例如 '|  |' 或 '   |'"""
    l_char = "|" if left_on else " "
    r_char = "|" if right_on else " "
    return l_char + (" " * s) + r_char


    
for line in sys.stdin:
if not line:
    continue
    
s_str, n_str = line.split()
s = int(s_str)

# 遇到 0 0 代表輸入結束
if s == 0 and n_str == '0':
    break
    
# 總高度為 2s + 3，由上往下一行一行印
for r in range(2 * s + 3):
    row_output = [] # 用來收集這一行所有數字的字串
    
    for digit in n_str:
        seg = segments[digit] # 取得該數字的 7 段顯示器開關
        
        if r == 0:
            # 頂部
            row_output.append(get_horizontal(seg[0], s))
        elif 1 <= r <= s:
            # 上半部直線
            row_output.append(get_vertical(seg[1], seg[2], s))
        elif r == s + 1:
            # 中間
            row_output.append(get_horizontal(seg[3], s))
        elif s + 2 <= r <= 2 * s + 1:
            # 下半部直線
            row_output.append(get_vertical(seg[4], seg[5], s))
        elif r == 2 * s + 2:
            # 底部
            row_output.append(get_horizontal(seg[6], s))
            
    # 將這一行各個數字拼接起來，數字與數字之間剛好用 1 個空格隔開
    print(" ".join(row_output))
    
# 題目要求：每個數字輸出後都要空一行
print()

# uva. 100 The 3n + 1 problem

    給你一個起始跟終點要求在其間3n+1演算法中次數最大的那一個
    Sample Input
    1 10
    100 200
    201 210
    900 1000
    Sample Output
    1 10 20
    100 200 125
    201 210 89
    900 1000 174

解題思路:

  • 要遍歷他給的數列因此3n+1可以先寫成一個函式方便呼叫
  • 記得他給的數字可能前大後小確保順序正確
  • 按照其要求輸入到EOF 輸出原始數字加最大值
  • python寫很容易tle因此換成java寫 但概念一模一樣

import java.util.*;
public class Main {
    public static int meth3_1(int num){
        int count=0;
        while(num!=1){
            if(num%2==0){
                num=num/2;
            }else{
                num=num*3+1;
            }
            count++;

        }
        return count;
    }
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (scanner.hasNextInt()) {
            int start = scanner.nextInt();
            int end = scanner.nextInt();
            int max = 0;
            int r_s = Math.min(start, end);
            int r_e = Math.max(start, end);
            for (int i = r_s; i <= r_e; i++) {
                int steps = meth3_1(i);
                if (steps > max) {
                    max = steps;
                }
            }
            System.out.printf("%d %d %d\n", start, end, max+1);

        }
        scanner.close();
    }
}
